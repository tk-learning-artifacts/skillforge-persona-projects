1. Sensor node

On the wall by the light switches, on 2 AAA batteries. It samples every second. Every minute it sends the average light, the loudest sound and the temperature. Every 10 seconds it checks for the "last one out" moment.

// Sensor node: micro:bit V2 on the wall. Sends 3 readings a minute.
const GATEWAY_SERIAL = 0   // set from the gateway page log. 0 accepts any sender.
const LIT = 70             // light above this: lights are on (measure your room)
const QUIET = 80           // sound below this: nobody talking (measure your room)
let mode = "remind"
let lastBlockPeak = 0

radio.setGroup(83)
radio.setTransmitSerialNumber(true)

radio.onReceivedString(function (cmd) {
    if (GATEWAY_SERIAL != 0 && radio.receivedPacket(RadioPacketProperty.SerialNumber) != GATEWAY_SERIAL) {
        return   // not our gateway
    }
    if (cmd == "remind" || cmd == "quiet") {
        mode = cmd
    } else if (cmd == "test") {
        basic.showIcon(IconNames.Yes)
        basic.pause(2000)
        basic.clearScreen()
    }   // anything else is ignored
})

basic.forever(function () {
    let lightSum = 0
    let minutePeak = 0
    for (let block = 0; block < 6; block++) {      // 6 blocks of 10 seconds
        let blockPeak = 0
        let blockLight = 0
        for (let s = 0; s < 10; s++) {
            let l = input.lightLevel()
            lightSum += l
            blockLight += l
            blockPeak = Math.max(blockPeak, input.soundLevel())
            basic.pause(1000)
        }
        // Last one out: loud, then quiet, with the lights still on.
        if (mode == "remind" && lastBlockPeak > QUIET + 20 && blockPeak < QUIET && blockLight / 10 > LIT) {
            music.playTone(523, music.beat(BeatFraction.Half))
            basic.showLeds(`
                . # # # .
                # . . . #
                . # . # .
                . . # . .
                . . # . .
                `)
            basic.pause(3000)
            basic.clearScreen()
        }
        lastBlockPeak = blockPeak
        minutePeak = Math.max(minutePeak, blockPeak)
    }
    radio.sendValue("light", Math.round(lightSum / 60))
    radio.sendValue("sound", minutePeak)
    radio.sendValue("temp", input.temperature())
})

input.onButtonPressed(Button.A, function () {
    basic.showNumber(control.deviceSerialNumber())   // handy during setup
})

The nudge pauses the loop for 3 seconds, so that minute is a little longer. The light is read while the screen is off, because the screen's LEDs do the sensing.

2. Gateway micro:bit

Plugged into the classroom PC by USB. It passes readings from the node to the PC as name:value lines. It passes modes from the PC to the node by radio.

// Gateway: micro:bit on the PC's USB port.
const NODE_SERIAL = 0   // set from the log. 0 forwards any sender and logs who it was.
radio.setGroup(83)
radio.setTransmitSerialNumber(true)
serial.setBaudRate(BaudRate.BaudRate115200)

radio.onReceivedValue(function (name, value) {
    let from = radio.receivedPacket(RadioPacketProperty.SerialNumber)
    if (NODE_SERIAL == 0) {
        serial.writeValue("from", from)
    } else if (from != NODE_SERIAL) {
        serial.writeValue("stranger", from)   // another class's node, or someone testing
        return
    }
    serial.writeValue(name, value)          // e.g. light:112
    led.toggle(4, 4)
})

serial.onDataReceived(serial.delimiters(Delimiters.NewLine), function () {
    let line = serial.readUntil(serial.delimiters(Delimiters.NewLine))
    if (line.indexOf("mode:remind") == 0) {
        radio.sendString("remind")
    } else if (line.indexOf("mode:quiet") == 0) {
        radio.sendString("quiet")
    } else if (line.indexOf("mode:test") == 0) {
        radio.sendString("test")
    }
})

input.onButtonPressed(Button.A, function () {
    basic.showNumber(control.deviceSerialNumber())
})

3. Gateway page

The gateway page runs in Chrome on the PC. What it does:

  • Opens the micro:bit's serial port at 115200 baud with Web Serial, after you click Connect.
  • Reads light, sound and temp lines, and groups them into minutes.
  • Every 5 minutes, builds one row: average light, loudest sound, average temperature, and lit, quiet minutes.
  • Posts it to the web app. The body is JSON, sent as plain text. That keeps it a "simple" request, so the browser sends no preflight check first.
  • Reads the reply. If the mode changed, writes mode:quiet (or remind, or test) back to the micro:bit.
  • Keeps failed rows in a queue in the browser's storage, and retries them oldest first.
  • Keeps the web app address and token in that browser only. They are never in the page's code.

The full code is gateway.js, with comments.

4. Apps Script web app

In the Sheet: Extensions, then Apps Script. Paste this. In Project Settings, add a Script Property called TOKEN with a long random value. Deploy as a web app that runs as you. Access must allow anonymous posts from the PC.

// Lights On, Nobody Home: takes a 5-minute row, replies with the mode.
const ROOMS = ['2E'];
const MODES = ['remind', 'quiet', 'test'];

function doPost(e) {
  const reply = (obj) => ContentService.createTextOutput(JSON.stringify(obj))
      .setMimeType(ContentService.MimeType.JSON);
  let row;
  try {
    row = JSON.parse(e.postData.contents);
  } catch (err) {
    return reply({ ok: false, error: 'not JSON' });
  }
  const token = PropertiesService.getScriptProperties().getProperty('TOKEN');
  if (!token || row.token !== token) return reply({ ok: false, error: 'bad token' });
  if (ROOMS.indexOf(row.room) < 0) return reply({ ok: false, error: 'unknown room' });

  // null is allowed: it means "no reading", and becomes a blank cell.
  const inRange = (v, lo, hi) => v === null || (typeof v === 'number' && v >= lo && v <= hi);
  if (!inRange(row.light, 0, 255) || !inRange(row.sound_peak, 0, 255) ||
      !inRange(row.temp_c, 0, 60) || !inRange(row.lit_quiet_min, 0, 5)) {
    return reply({ ok: false, error: 'value out of range' });
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const book = SpreadsheetApp.getActiveSpreadsheet();
    const data = book.getSheetByName('Data');
    const from = Math.max(2, data.getLastRow() - 199);
    const recent = data.getRange(from, 1, 200, 1).getValues().flat();
    if (recent.indexOf(row.id) < 0) {   // a retry never makes a double row
      data.appendRow([row.id, row.time, new Date(), row.room, row.light,
        row.sound_peak, row.temp_c, row.lit_quiet_min, String(row.status || 'ok').slice(0, 20)]);
    }
    let mode = String(book.getSheetByName('Control').getRange('B1').getValue()).trim().toLowerCase();
    if (MODES.indexOf(mode) < 0) mode = 'quiet';   // a typo in B1 means no beeps
    return reply({ ok: true, mode: mode });
  } finally {
    lock.releaseLock();
  }
}

A reply looks like {"ok":true,"mode":"remind"}. Apps Script sends replies through a redirect to script.googleusercontent.com. The browser follows it by itself. If you test with curl, add -L.

Notice the numbers have no quotes. If a student's code sends "light": "112", the script refuses it: a string is not a number. Control tab: A1 says "mode", B1 holds remind, quiet or test. Protect the tab so only the eco monitors and the teacher can edit it.
