def startLevel(n: number):
    global level
    level = n
    sprites.destroy_all_sprites_of_kind(SpriteKind.food)
    sprites.destroy_all_sprites_of_kind(SpriteKind.enemy)
    sprites.destroy_all_sprites_of_kind(SpriteKind.Goal)
    tiles.set_current_tilemap(LEVELS[n])
    tiles.place_on_random_tile(bot, DOOR)
    for spot in tiles.get_tiles_by_type(CUPSPOT):
        cup = sprites.create(img("""

def on_on_overlap(sprite, otherSprite):
    info.change_life_by(-1)
    tiles.place_on_random_tile(sprite, DOOR)
sprites.on_overlap(SpriteKind.player, SpriteKind.enemy, on_on_overlap)

def on_on_update():
    if tiles.tile_at_location_equals(bot.tilemap_location(), WATER):
        info.change_life_by(-1)
        tiles.place_on_random_tile(bot, DOOR)
game.on_update(on_on_update)
