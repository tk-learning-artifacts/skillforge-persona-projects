# Found a duckling: it follows you home.

def on_on_overlap(sprite, otherSprite):
    global ducklingsLeft
    otherSprite.set_kind(SpriteKind.Found)
    otherSprite.set_flag(SpriteFlag.GHOST_THROUGH_WALLS, True)
    otherSprite.follow(sprite, 50)
    info.change_score_by(1)
    ducklingsLeft += -1
    music.play(music.melody_playable(music.ba_ding),
        music.PlaybackMode.IN_BACKGROUND)
    if ducklingsLeft == 0:
        openGate()
sprites.on_overlap(SpriteKind.player, SpriteKind.food, on_on_overlap)

# The gate opens once every duckling is found.
def openGate():
    for spot5 in tiles.get_tiles_by_type(GATE):
        tiles.set_wall_at(spot5, False)
        tiles.set_tile_at(spot5, GRASS)
    mama.say_text("Gate open!", 1500)
    music.play(music.melody_playable(music.power_up),
        music.PlaybackMode.IN_BACKGROUND)
# A mower sends you back to the nest and costs a life.

def on_on_overlap2(sprite2, otherSprite2):
    info.change_life_by(-1)
    scene.camera_shake(3, 300)
    music.play(music.melody_playable(music.small_crash),
        music.PlaybackMode.IN_BACKGROUND)
    tiles.place_on_random_tile(sprite2, NEST)
sprites.on_overlap(SpriteKind.player, SpriteKind.enemy, on_on_overlap2)

# The pond ends park 1. In park 2 it wins the game.

def on_on_overlap3(sprite3, otherSprite3):
    if level == 0:
        startLevel(1)
    else:
        game.over(True)
sprites.on_overlap(SpriteKind.player, SpriteKind.Goal, on_on_overlap3)

startLevel(0)
