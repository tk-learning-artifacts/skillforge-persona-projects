One answer. It uses the same idea as to_denary(), but takes each place away.

def to_binary(number):
    bits = ""
    place = 8
    while place >= 1:
        if number >= place:
            bits = bits + "1"
            number = number - place
        else:
            bits = bits + "0"
        place = place // 2
    return bits


number = 11
print("The keypad shows", number)
guess = input("Type it as 4 bits: ")
while guess != to_binary(number):
    print("Beep. Try again.")
    guess = input("Type it as 4 bits: ")
print("Click. The door opens.")
