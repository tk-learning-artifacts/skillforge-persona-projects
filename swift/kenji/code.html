Code and tests

Every Swift file in PigOut.swiftpm.zip, then the tests.

Rules apart from screens

PigGame is a plain struct. It knows the rules and nothing about SwiftUI. roll and hold change the game and return an event, like .pigOut. GameView keeps one PigGame in @State and turns each event into words and motion. The rules stand alone, so main.swift can test them on a Mac with no screen.

Read in this order: PigGame.swift, GameView.swift, then the rest.

Every file was checked with the Swift 6.3 compiler on a Mac. Two lines only work on iPad and iPhone, so the check skipped them. All ten tests pass. Package.swift is left out here. Swift Playgrounds writes it for you.

PigGame.swift

The rules. Look at roll: a 1 clears the turn and passes the die. Look at botWantsToRoll and its comment. That is the hold at 20 maths in four lines.

68 lines.

// The rules of Pig, with no SwiftUI in sight.
// Keeping the rules apart from the screens means I can test them on their own.

enum Player: Int {
    case one = 0, two = 1
    var other: Player { self == .one ? .two : .one }
}

// What just happened. The screen turns each event into words, a haptic and a wiggle.
enum PigEvent: Equatable {
    case rolled(Int)
    case pigOut          // rolled a 1: turn points lost
    case held(Int)       // banked this many points
    case won(Player)
}

struct PigGame {
    let target: Int
    private(set) var scores = [0, 0]
    private(set) var turnTotal = 0
    private(set) var current = Player.one
    private(set) var lastRoll: Int? = nil
    private(set) var winner: Player? = nil

    init(target: Int = 50) {
        self.target = target
    }

    func score(of player: Player) -> Int {
        scores[player.rawValue]
    }

    // Roll the die. Pass in the value so tests can choose it.
    mutating func roll(_ value: Int = Int.random(in: 1...6)) -> PigEvent {
        guard winner == nil else { return .won(winner!) }
        lastRoll = value
        if value == 1 {
            turnTotal = 0
            current = current.other
            return .pigOut
        }
        turnTotal += value
        return .rolled(value)
    }

    // Bank the turn total and pass the die.
    mutating func hold() -> PigEvent {
        guard winner == nil else { return .won(winner!) }
        let banked = turnTotal
        scores[current.rawValue] += banked
        turnTotal = 0
        if scores[current.rawValue] >= target {
            winner = current
            return .won(current)
        }
        current = current.other
        return .held(banked)
    }
}

// Porkchop, the bot, holds at 20, or as soon as holding would win.
// Why 20? One more roll gains about 4 points five times in six,
// and loses the whole turn total one time in six.
// Those balance when the turn total is 20.
func botWantsToRoll(turnTotal: Int, score: Int, target: Int) -> Bool {
    if score + turnTotal >= target { return false }
    return turnTotal < 20
}

GameView.swift

import SwiftUI

// Screen 2: the game.
struct GameView: View {
    let versusBot: Bool
    @State private var game: PigGame
    @State private var message: String
    @State private var spin = 0.0        // degrees the die has turned
    @State private var pigOuts = 0       // goes up by 1 on every pig out, to shake the die
    @State private var rolls = 0         // goes up by 1 on every roll, for the haptic tap
    @State private var botBusy = false
    @Environment(\.accessibilityReduceMotion) private var reduceMotion

    init(versusBot: Bool, target: Int) {
        self.versusBot = versusBot
        _game = State(initialValue: PigGame(target: target))
        _message = State(initialValue: versusBot ? "Your turn. Roll the die." : "Player 1's turn. Roll the die.")
    }

    var body: some View {
        VStack(spacing: 20) {
            HStack(spacing: 12) {
                ScoreCard(name: name(.one), score: game.score(of: .one),
                          target: game.target, active: game.current == .one)
                ScoreCard(name: name(.two), score: game.score(of: .two),
                          target: game.target, active: game.current == .two)
            }

            DieView(value: game.lastRoll ?? 6, size: 140)
                .rotationEffect(.degrees(spin))
                .modifier(Shake(times: pigOuts))
                .opacity(game.lastRoll == nil ? 0.35 : 1)

            Text("Turn: \(game.turnTotal)")
                .font(.system(.largeTitle, design: .rounded, weight: .heavy))
                .contentTransition(.numericText(value: Double(game.turnTotal)))

            // Every event is said in words, not only by sound or motion.
            Text(message)
                .font(.title3)
                .multilineTextAlignment(.center)
                .frame(minHeight: 60)

            if game.winner != nil {
                Button("Play again") {
                    game = PigGame(target: game.target)
                    message = "\(turnOf(.one)) Roll the die."
                }
                .buttonStyle(.borderedProminent)
                .font(.title2.weight(.bold))
            } else {
                HStack(spacing: 16) {
                    Button { roll() } label: {
                        Label("Roll", systemImage: "die.face.5")
                            .frame(maxWidth: .infinity, minHeight: 56)
                    }
                    .buttonStyle(.borderedProminent)

                    Button { hold() } label: {
                        Label("Hold", systemImage: "tray.and.arrow.down")
                            .frame(maxWidth: .infinity, minHeight: 56)
                    }
                    .buttonStyle(.bordered)
                    .disabled(game.turnTotal == 0)
                }
                .font(.title2.weight(.bold))
                .disabled(botsTurn || botBusy)
            }
            Spacer()
        }
        .padding()
        .navigationTitle("First to \(game.target)")
        .navigationBarTitleDisplayMode(.inline)
        .sensoryFeedback(.impact, trigger: rolls)
        .sensoryFeedback(.error, trigger: pigOuts)
    }

    private var botsTurn: Bool { versusBot && game.current == .two }

    private func name(_ player: Player) -> String {
        if player == .one { return versusBot ? "You" : "Player 1" }
        return versusBot ? "Porkchop" : "Player 2"
    }

    // "Your turn." or "Porkchop's turn."
    private func turnOf(_ player: Player) -> String {
        name(player) == "You" ? "Your turn." : "\(name(player))'s turn."
    }

    private func roll() {
        let player = game.current
        report(game.roll(), by: player)
        if botsTurn { playBot() }
    }

    private func hold() {
        let player = game.current
        report(game.hold(), by: player)
        if botsTurn { playBot() }
    }

    // Turn an event into words, motion, a haptic and a VoiceOver announcement.
    private func report(_ event: PigEvent, by player: Player) {
        let who = name(player)
        switch event {
        case .rolled(let value):
            rolls += 1
            animate(.easeOut(duration: 0.35)) { spin += 360 }
            message = "\(who) rolled \(value). Turn total \(game.turnTotal)."
        case .pigOut:
            rolls += 1
            animate(.linear(duration: 0.4)) { pigOuts += 1 }
            message = "Pig out! \(who) rolled a 1 and lost the turn. \(turnOf(game.current))"
        case .held(let banked):
            message = "\(who) banked \(banked). \(turnOf(game.current))"
        case .won(let winner):
            let verb = name(winner) == "You" ? "win" : "wins"
            message = "\(name(winner)) \(verb), \(game.score(of: winner)) to \(game.score(of: winner.other))!"
        }
        AccessibilityNotification.Announcement(message).post()
    }

    // With Reduce Motion on, change the value with no animation.
    private func animate(_ animation: Animation, _ change: () -> Void) {
        if reduceMotion {
            change()
        } else {
            withAnimation(animation, change)
        }
    }

    // Porkchop plays slowly, so you can watch every roll.
    private func playBot() {
        guard !botBusy else { return }
        botBusy = true
        Task {
            while botsTurn && game.winner == nil {
                try? await Task.sleep(for: .seconds(1.2))
                let wantsToRoll = botWantsToRoll(turnTotal: game.turnTotal,
                                                 score: game.score(of: .two),
                                                 target: game.target)
                report(wantsToRoll ? game.roll() : game.hold(), by: .two)
            }
            botBusy = false
        }
    }
}

// One player's score, with a bar that fills towards the target.
struct ScoreCard: View {
    let name: String
    let score: Int
    let target: Int
    let active: Bool

    var body: some View {
        VStack(spacing: 6) {
            Text(active ? "▶ \(name)" : name)
                .font(.headline)
            Text("\(score)")
                .font(.system(.largeTitle, design: .rounded, weight: .bold))
                .contentTransition(.numericText(value: Double(score)))
            ProgressView(value: Double(min(score, target)), total: Double(target))
        }
        .padding()
        .frame(maxWidth: .infinity)
        .background(active ? Color.pigPink.opacity(0.15) : Color.gray.opacity(0.1),
                    in: RoundedRectangle(cornerRadius: 16))
        .overlay(RoundedRectangle(cornerRadius: 16)
            .stroke(active ? Color.pigPink : .clear, lineWidth: 3))
        // The arrow and the outline show whose turn it is, so colour is not the only clue.
        .accessibilityElement(children: .ignore)
        .accessibilityLabel("\(name), \(score) of \(target)\(active ? ", playing now" : "")")
    }
}

// Shakes a view side to side, three times, whenever `times` goes up by 1.
struct Shake: GeometryEffect {
    var animatableData: CGFloat

    init(times: Int) {
        animatableData = CGFloat(times)
    }

    func effectValue(size: CGSize) -> ProjectionTransform {
        ProjectionTransform(CGAffineTransform(translationX: 12 * sin(animatableData * .pi * 6), y: 0))
    }
}

import SwiftUI

// Screen 1: the title, the rules, and who you play against.
struct TitleView: View {
    @State private var versusBot = true
    @State private var target = 50

    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 24) {
                    DieView(value: 5, size: 96)
                        .accessibilityHidden(true)
                    Text("Pig Out")
                        .font(.system(size: 56, weight: .black, design: .rounded))
                        .foregroundStyle(Color.pigPink)

                    VStack(alignment: .leading, spacing: 10) {
                        Text("How to play").font(.headline)
                        Label("Roll as often as you like. Each roll adds to your turn.", systemImage: "die.face.3")
                        Label("Roll a 1 and you lose the whole turn. Pig out!", systemImage: "exclamationmark.triangle")
                        Label("Hold to bank your turn and pass the die.", systemImage: "tray.and.arrow.down")
                        Label("First to \(target) wins.", systemImage: "flag.checkered")
                    }
                    .frame(maxWidth: .infinity, alignment: .leading)
                    .padding()
                    .background(.thinMaterial, in: RoundedRectangle(cornerRadius: 20))

                    Picker("Play against", selection: $versusBot) {
                        Text("Porkchop the bot").tag(true)
                        Text("A friend").tag(false)
                    }
                    .pickerStyle(.segmented)

                    Stepper("First to \(target)", value: $target, in: 20...100, step: 10)

                    NavigationLink {
                        GameView(versusBot: versusBot, target: target)
                    } label: {
                        Text("Play")
                            .font(.title2.weight(.bold))
                            .frame(maxWidth: .infinity, minHeight: 56)
                    }
                    .buttonStyle(.borderedProminent)
                }
                .padding()
            }
            .navigationTitle("Pig Out")
            .toolbar(.hidden, for: .navigationBar)
        }
        // Every button, bar and back arrow in the app uses the same pink.
        .tint(Color.pigPink)
    }
}

extension Color {
    // White text on this pink has a contrast of about 5.6 to 1.
    static let pigPink = Color(red: 0.75, green: 0.16, blue: 0.40)
}

DieView.swift

A die face from nine circles on a grid. The pips that are not used are see-through. A 1 has a pink pip, so a pig out stands out.

47 lines.

import SwiftUI

// A die face drawn with circles, so it needs no images.
struct DieView: View {
    let value: Int
    var size: CGFloat = 120

    // Where the pips go on a 3 by 3 grid, counted 0 to 8.
    private var pips: [Int] {
        switch value {
        case 1: [4]
        case 2: [0, 8]
        case 3: [0, 4, 8]
        case 4: [0, 2, 6, 8]
        case 5: [0, 2, 4, 6, 8]
        default: [0, 2, 3, 5, 6, 8]
        }
    }

    var body: some View {
        let pip = size / 6
        RoundedRectangle(cornerRadius: size / 5)
            .fill(.white)
            .stroke(Color.black.opacity(0.8), lineWidth: 3)
            .frame(width: size, height: size)
            .overlay {
                Grid(horizontalSpacing: pip / 2, verticalSpacing: pip / 2) {
                    ForEach(0..<3, id: \.self) { row in
                        GridRow {
                            ForEach(0..<3, id: \.self) { col in
                                Circle()
                                    .fill(value == 1 ? Color.pigPink : .black)
                                    .frame(width: pip, height: pip)
                                    .opacity(pips.contains(row * 3 + col) ? 1 : 0)
                            }
                        }
                    }
                }
            }
            .accessibilityElement()
            .accessibilityLabel("Die showing \(value)")
    }
}

// Tests for the Pig rules. Run them on a Mac with:
// swiftc PigGame.swift main.swift -o pigtests && ./pigtests
var failures = 0
func check(_ ok: Bool, _ what: String) {
    print(ok ? "PASS" : "FAIL", what)
    if !ok { failures += 1 }
}

var g = PigGame(target: 50)
_ = g.roll(4); _ = g.roll(6)
check(g.turnTotal == 10, "two rolls add up: 4 + 6 = 10")
check(g.hold() == .held(10) && g.score(of: .one) == 10, "hold banks 10 for player 1")
check(g.current == .two && g.turnTotal == 0, "hold passes the die and clears the turn")
_ = g.roll(5)
check(g.roll(1) == .pigOut && g.turnTotal == 0, "a 1 wipes the turn total")
check(g.current == .one && g.score(of: .two) == 0, "a 1 passes the die and banks nothing")
var w = PigGame(target: 20)
_ = w.roll(6); _ = w.roll(6); _ = w.roll(6); _ = w.roll(2)
check(w.hold() == .won(.one) && w.winner == .one, "reaching the target wins")
check(w.roll(3) == .won(.one) && w.turnTotal == 0, "no more rolls after a win")
check(!botWantsToRoll(turnTotal: 20, score: 0, target: 50), "bot holds at 20")
check(botWantsToRoll(turnTotal: 19, score: 0, target: 50), "bot rolls at 19")
check(!botWantsToRoll(turnTotal: 6, score: 45, target: 50), "bot holds when holding wins")

// Play 100,000 bot turns and average the points banked per turn.
var total = 0
let turns = 100_000
for _ in 0..<turns {
    var turn = PigGame(target: 1_000)
    while botWantsToRoll(turnTotal: turn.turnTotal, score: 0, target: 1_000) {
        if turn.roll() == .pigOut { break }
    }
    if turn.current == .one { total += turn.turnTotal }
}
print("Hold at 20: average", Double(total) / Double(turns), "points a turn over", turns, "turns")
print(failures == 0 ? "All tests passed." : "\(failures) failed.")
