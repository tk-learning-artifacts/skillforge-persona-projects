The code

Every Swift file in TeamPoints.swiftpm.zip, with a note on what to look at.

Where the state lives

There is one list of teams. It lives in TeamsView, marked @State. TeamView never owns a score. It gets a @Binding to one team, so when it changes the score, it changes the list's copy. That is why the score is still there when you tap Back.

Read the files in this order: Team.swift, TeamsView.swift, TeamView.swift.

I checked that every file builds against Apple's iOS SDK with Swift 6.3. Package.swift is left out here. Swift Playgrounds writes it for you.

import SwiftUI

// Screen 1: every team and its score.
// The scores live HERE, in one @State array.
// Screen 2 gets a @Binding to one team, so its changes come back here.
struct TeamsView: View {
    @State private var teams = Team.starters
    @State private var askingToReset = false

    var body: some View {
        NavigationStack {
            List {
                Section {
                    // $teams gives me a binding to each team.
                    ForEach($teams) { $team in
                        NavigationLink {
                            TeamView(team: $team)
                        } label: {
                            TeamRow(team: team, isLeading: isLeading(team))
                        }
                    }
                } footer: {
                    Text("Tap a team to give points.")
                }
            }
            .navigationTitle("Team Points")
            .toolbar {
                Button("Reset") {
                    askingToReset = true
                }
                .disabled(teams.allSatisfy { $0.points == 0 })
            }
            .confirmationDialog("Set every score back to 0?",
                                isPresented: $askingToReset,
                                titleVisibility: .visible) {
                Button("Reset all scores", role: .destructive) {
                    for index in teams.indices {
                        teams[index].points = 0
                    }
                }
            }
        }
        // Toolbar text and the back button: black in light mode, white in dark mode.
        // Orange was hard to read on white.
        .tint(.primary)
    }

    // A team leads if it has the top score and that score is above 0.
    func isLeading(_ team: Team) -> Bool {
        let top = teams.map(\.points).max() ?? 0
        return top > 0 && team.points == top
    }
}

// One row in the list: shape, name, a crown for the leader, and the score.
struct TeamRow: View {
    let team: Team
    let isLeading: Bool

    var body: some View {
        HStack(spacing: 12) {
            Image(systemName: team.symbol)
                .font(.title2)
                .foregroundStyle(team.color)
                .frame(width: 32)
            Text(team.name)
                .font(.title3.weight(.semibold))
            if isLeading {
                Image(systemName: "crown.fill")
                    .foregroundStyle(.orange)
            }
            Spacer()
            Text("\(team.points)")
                .font(.system(.title, design: .rounded, weight: .bold))
                .monospacedDigit()
        }
        .padding(.vertical, 6)
        // VoiceOver reads the whole row as one sentence.
        .accessibilityElement(children: .ignore)
        .accessibilityLabel("\(team.name), \(team.points) points\(isLeading ? ", in the lead" : "")")
    }
}

import SwiftUI

// Screen 2: one team's score, with buttons to change it.
// @Binding means this screen changes the team that lives in TeamsView.
struct TeamView: View {
    @Binding var team: Team
    @Environment(\.accessibilityReduceMotion) private var reduceMotion
    @State private var bump = false

    var body: some View {
        VStack(spacing: 28) {
            Image(systemName: team.symbol)
                .font(.system(size: 56))
                .foregroundStyle(team.color)
                .accessibilityHidden(true)

            // The big score. It rolls to the new number and pops a little.
            Text("\(team.points)")
                .font(.system(size: 120, weight: .heavy, design: .rounded))
                .monospacedDigit()
                .contentTransition(.numericText(value: Double(team.points)))
                .scaleEffect(bump ? 1.15 : 1)
                .accessibilityLabel("\(team.points) points")

            HStack(spacing: 16) {
                PointButton(label: "−1", color: .gray) { change(by: -1) }
                    .accessibilityLabel("Take away 1 point")
                    .disabled(team.points == 0)
                PointButton(label: "+1", color: team.color) { change(by: 1) }
                    .accessibilityLabel("Add 1 point")
                PointButton(label: "+3", color: team.color) { change(by: 3) }
                    .accessibilityLabel("Add 3 points")
            }

            // Renaming here renames the team on screen 1 too, because of @Binding.
            TextField("Team name", text: $team.name)
                .textFieldStyle(.roundedBorder)
                .font(.title3)
                .multilineTextAlignment(.center)
                .frame(maxWidth: 280)
                .accessibilityLabel("Team name")

            Spacer()
        }
        .padding(.top, 32)
        .padding(.horizontal)
        .navigationTitle(team.name)
        .navigationBarTitleDisplayMode(.inline)
    }

    func change(by amount: Int) {
        let newPoints = max(0, team.points + amount)
        if reduceMotion {
            // No rolling numbers or popping for people who turned on Reduce Motion.
            team.points = newPoints
            return
        }
        withAnimation(.spring(duration: 0.3)) {
            team.points = newPoints
            bump = true
        }
        withAnimation(.spring(duration: 0.3).delay(0.15)) {
            bump = false
        }
    }
}

// A big round-cornered button. 80 by 64 points, well over Apple's 44 by 44.
struct PointButton: View {
    let label: String
    let color: Color
    let action: () -> Void

    var body: some View {
        Button(action: action) {
            Text(label)
                .font(.system(.title, design: .rounded, weight: .bold))
                .frame(width: 80, height: 64)
                .foregroundStyle(.white)
                .background(color, in: RoundedRectangle(cornerRadius: 16))
        }
    }
}
