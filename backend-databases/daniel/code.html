The code

Four files. I tested them with Flask 3.1 on Python 3.13. Download all four (zip).

  • schema.sql makes the three tables.
  • setup_db.py is the database setup script. It runs schema.sql, adds made-up sign-ups and one teacher login.
  • app.py is the Flask app. Five routes.
  • templates/index.html is the one page. Flask fills it in.

Run it on a computer with Python

  1. Unzip the folder. Open a terminal in it.
  2. pip install flask
  3. python setup_db.py. It asks for a teacher username and password. Make them up.
  4. flask --app app run --debug, then open http://127.0.0.1:5000.

schema.sql

Three tables. The comment says what I took out, and why.

schema.sql
-- Trial Day sign-up tables. setup_db.py runs this file.
DROP TABLE IF EXISTS signups;
DROP TABLE IF EXISTS slots;
DROP TABLE IF EXISTS teachers;

-- One row for each trial session a CCA runs.
CREATE TABLE slots (
  id INTEGER PRIMARY KEY,
  cca TEXT NOT NULL,
  time_label TEXT NOT NULL,
  places INTEGER NOT NULL
);

-- One row for each student who signs up.
-- I took out phone number and parent email. The teacher meets them at the session anyway.
CREATE TABLE signups (
  id INTEGER PRIMARY KEY,
  slot_id INTEGER NOT NULL REFERENCES slots(id),
  first_name TEXT NOT NULL,
  form_class TEXT NOT NULL
);

-- Only teachers have logins. Students sign up without one.
CREATE TABLE teachers (
  id INTEGER PRIMARY KEY,
  username TEXT NOT NULL UNIQUE,
  pw_hash TEXT NOT NULL
);

setup_db.py: the setup script

It uses input and getpass, so the teacher password never sits in the code. The last lines make a secret key for Flask's session cookie. Flask needs one before logins work.

setup_db.py
# Makes trial.db from schema.sql and puts in made-up data.
# Run it in a Bash console:  python3 setup_db.py
# Careful: running it again wipes everything.
import getpass
import os
import secrets
import sqlite3

from werkzeug.security import generate_password_hash

folder = os.path.dirname(os.path.abspath(__file__))
con = sqlite3.connect(os.path.join(folder, "trial.db"))

with open(os.path.join(folder, "schema.sql")) as f:
    con.executescript(f.read())

slots = [
    ("Robotics", "Wed 2:30 pm", 6),
    ("Robotics", "Fri 2:30 pm", 6),
    ("Infocomm", "Wed 2:30 pm", 8),
    ("Media Club", "Thu 3:00 pm", 5),
]
con.executemany("INSERT INTO slots (cca, time_label, places) VALUES (?, ?, ?)", slots)

# Made-up students. Never put real names in here.
fake_students = [
    (1, "Ali", "1E1"),
    (1, "Bee Lin", "1E3"),
    (3, "Chandra", "1E2"),
    (4, "Dev", "1E4"),
]
con.executemany(
    "INSERT INTO signups (slot_id, first_name, form_class) VALUES (?, ?, ?)",
    fake_students,
)

# The password is typed in, not written in this file.
username = input("Teacher username: ").strip().lower()
password = getpass.getpass("Teacher password (8 or more characters): ")
if len(password) < 8:
    raise SystemExit("Too short. Run me again.")
con.execute(
    "INSERT INTO teachers (username, pw_hash) VALUES (?, ?)",
    (username, generate_password_hash(password)),
)

con.commit()  # I forgot this line once. Nothing saved.
con.close()

with open(os.path.join(folder, "secret.key"), "w") as f:
    f.write(secrets.token_hex())

print("Done. Made trial.db and secret.key.")

app.py

The part I had to read three times was the LEFT JOIN with COUNT. A plain JOIN drops a session with no sign-ups at all. So Robotics on Friday vanished from my list until I changed it.

app.py
# Trial Day sign-up: Sec 1 students pick a CCA trial session.
# One page, made by Flask from templates/index.html.
import os
import re
import sqlite3

from flask import Flask, flash, redirect, render_template, request, session
from werkzeug.security import check_password_hash

folder = os.path.dirname(os.path.abspath(__file__))
DB_FILE = os.path.join(folder, "trial.db")

app = Flask(__name__)
with open(os.path.join(folder, "secret.key")) as f:
    app.secret_key = f.read().strip()


def get_db():
    con = sqlite3.connect(DB_FILE)
    con.row_factory = sqlite3.Row
    return con


@app.get("/")
def home():
    con = get_db()
    # LEFT JOIN, so a slot with no sign-ups still shows up, with 0 taken.
    slots = con.execute(
        "SELECT slots.id, slots.cca, slots.time_label, slots.places, "
        "COUNT(signups.id) AS taken "
        "FROM slots LEFT JOIN signups ON signups.slot_id = slots.id "
        "GROUP BY slots.id ORDER BY slots.cca, slots.id"
    ).fetchall()
    signups = []
    if "teacher" in session:
        signups = con.execute(
            "SELECT signups.id, signups.first_name, signups.form_class, "
            "slots.cca, slots.time_label "
            "FROM signups JOIN slots ON slots.id = signups.slot_id "
            "ORDER BY slots.cca, slots.id, signups.form_class"
        ).fetchall()
    con.close()
    return render_template("index.html", slots=slots, signups=signups)


@app.post("/signup")
def signup():
    first_name = request.form.get("first_name", "").strip()
    form_class = request.form.get("form_class", "").strip().upper()
    slot_id = request.form.get("slot_id", "")
    if not 1 <= len(first_name) <= 30:
        flash("Type a first name, 30 letters at most.")
        return redirect("/")
    if not re.fullmatch(r"[1-5][A-Z]{0,2}[0-9]{1,2}", form_class):
        flash("Type your class like 1E2.")
        return redirect("/")

    con = get_db()
    slot = con.execute(
        "SELECT slots.cca, slots.time_label, slots.places, COUNT(signups.id) AS taken "
        "FROM slots LEFT JOIN signups ON signups.slot_id = slots.id "
        "WHERE slots.id = ? GROUP BY slots.id",
        (slot_id,),
    ).fetchone()
    if slot is None:
        flash("Pick a session.")
    elif slot["taken"] >= slot["places"]:
        flash(f"Sorry, {slot['cca']} on {slot['time_label']} is full.")
    else:
        con.execute(
            "INSERT INTO signups (slot_id, first_name, form_class) VALUES (?, ?, ?)",
            (slot_id, first_name, form_class),
        )
        con.commit()
        flash(f"Done! See you at {slot['cca']}, {slot['time_label']}.")
    con.close()
    return redirect("/")  # back to the page, so a refresh won't sign up twice


@app.post("/login")
def login():
    username = request.form.get("username", "").strip().lower()
    password = request.form.get("password", "")
    con = get_db()
    teacher = con.execute(
        "SELECT username, pw_hash FROM teachers WHERE username = ?", (username,)
    ).fetchone()
    con.close()
    if teacher and check_password_hash(teacher["pw_hash"], password):
        session["teacher"] = teacher["username"]
        flash("Logged in. The sign-up list is at the bottom.")
    else:
        flash("Wrong username or password.")
    return redirect("/")


@app.post("/logout")
def logout():
    session.clear()
    flash("Logged out.")
    return redirect("/")


@app.post("/remove/<int:signup_id>")
def remove(signup_id):
    if "teacher" not in session:
        return "Teachers only.", 403
    con = get_db()
    con.execute("DELETE FROM signups WHERE id = ?", (signup_id,))
    con.commit()
    con.close()
    flash("Removed.")
    return redirect("/")

How the copy on my app page is different

  • It is JavaScript, in server.js. The routes and SQL are copied from app.py. I checked them side by side.
  • Werkzeug's default hash is scrypt, and browsers can't do scrypt. So the page uses werkzeug's pbkdf2 format instead. Python's check_password_hash reads both.
  • The session lives in your browser's storage, not a signed cookie.
  • Version 1 lives only on the page. Its table, teachers_v1, holds a plain password on purpose.
