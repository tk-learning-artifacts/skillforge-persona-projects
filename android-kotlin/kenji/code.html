package com.example.hitbox

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.activity.enableEdgeToEdge
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.saveable.Saver
import androidx.compose.runtime.saveable.rememberSaveable
import androidx.compose.runtime.setValue
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.rememberNavController
import androidx.navigation.toRoute
import kotlinx.serialization.Serializable

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent {
            HitboxTheme {
                HitboxApp()
            }
        }
    }
}

// The two screens. Navigation Compose uses these as routes.
@Serializable
object WordList

@Serializable
data class WordDetail(val id: Int)

// A Set can't go in a Bundle, so save the ids as an IntArray.
private val KnownSaver = Saver<Set<Int>, IntArray>(
    save = { it.toIntArray() },
    restore = { it.toSet() }
)

@Composable
fun HitboxApp() {
    val navController = rememberNavController()

    // The words you know. It lives here, above both screens, so both can see it.
    // rememberSaveable keeps it when you rotate the phone.
    var known by rememberSaveable(stateSaver = KnownSaver) { mutableStateOf(emptySet<Int>()) }

    NavHost(navController = navController, startDestination = WordList) {
        composable<WordList> {
            WordListScreen(
                words = gameWords,
                known = known,
                onWordClick = { word -> navController.navigate(WordDetail(word.id)) }
            )
        }
        composable<WordDetail> { entry ->
            val id = entry.toRoute<WordDetail>().id
            val word = gameWords.first { it.id == id }
            WordDetailScreen(
                word = word,
                isKnown = id in known,
                onKnownChange = { nowKnown -> known = if (nowKnown) known + id else known - id },
                onBack = { navController.popBackStack() }
            )
        }
    }
}

import androidx.compose.ui.Modifier
import androidx.compose.ui.tooling.preview.Preview
import androidx.compose.ui.unit.dp

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun WordDetailScreen(
    word: Word,
    isKnown: Boolean,
    onKnownChange: (Boolean) -> Unit,
    onBack: () -> Unit
) {
    // Only this screen needs to know if the meaning is showing,
    // so this state stays here and is not hoisted.
    var revealed by rememberSaveable { mutableStateOf(false) }

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Level ${word.level}") },
                navigationIcon = { TextButton(onClick = onBack) { Text("< Back") } }
            )
        }
    ) { inner ->
        Column(
            modifier = Modifier
                .padding(inner)
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            Text(word.term, style = MaterialTheme.typography.displaySmall)

            if (revealed) {
                Card(Modifier.fillMaxWidth()) {
                    Column(Modifier.padding(16.dp), verticalArrangement = Arrangement.spacedBy(8.dp)) {
                        Text(word.meaning, style = MaterialTheme.typography.titleLarge)
                        Text("In a game: ${word.example}")
                    }
                }
                Text("Did you know it?")
                Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                    Button(onClick = { onKnownChange(true); onBack() }) { Text("I knew it") }
                    OutlinedButton(onClick = { onKnownChange(false); onBack() }) { Text("Not yet") }
                }
            } else {
                Text("Say the meaning out loud first. Then check.")
                Button(onClick = { revealed = true }) { Text("Show meaning") }
            }

            if (isKnown) {
                Text("★ You marked this one as known.", color = MaterialTheme.colorScheme.primary)
            }
        }
    }
}

import androidx.compose.foundation.clickable

import androidx.compose.material3.MaterialTheme

import androidx.compose.material3.TopAppBar
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.tooling.preview.Preview
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.ui.unit.dp

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun WordListScreen(
    words: List<Word>,
    known: Set<Int>,
    onWordClick: (Word) -> Unit
) {
    Scaffold(
        topBar = {
            TopAppBar(title = { Text("Hitbox: game words") })
        }
    ) { inner ->
        Column(Modifier.padding(inner)) {
            // State goes down: the list only reads known. It never changes it.
            Text(
                text = "Known ${known.size} of ${words.size}",
                style = MaterialTheme.typography.titleMedium,
                modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp)
            )
            LinearProgressIndicator(
                progress = { known.size / words.size.toFloat() },
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp)
            )
            // LazyColumn only builds the rows on screen.
            LazyColumn {
                items(words, key = { it.id }) { word ->
                    ListItem(
                        headlineContent = { Text(word.term, fontWeight = FontWeight.Bold) },
                        supportingContent = { Text(word.meaning) },
                        overlineContent = { Text("Level ${word.level}") },
                        trailingContent = { Text(if (word.id in known) "★" else "☆") },
                        // Events go up: tell the parent which word was tapped.
                        modifier = Modifier.clickable { onWordClick(word) }
                    )
                    HorizontalDivider()
                }
            }
        }
    }
}
