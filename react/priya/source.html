Every file in the Checkpoint project, with what to notice in each. Read App.jsx first. It holds the state and the effect.

Download the project (.zip) See the folder on GitHub

To run it: unzip, then npm install and npm run dev, with Node.js 20.19 or newer. Or put the files in a GitHub repository and open it in Codespaces. The built app is in app/.

How the parts fit

State lives in App, the closest parent of every part that needs it. Data goes down as props. Clicks come back up through functions like onPick.

  • App: state for the set, the quiz, an error, retries, the question number and the answers. The fetch effect.
    • SetPicker: props sets, current, onPick. A list of buttons.
    • Progress: props questions, answers, index. A list of dots.
    • QuestionCard: props question, number, total, outcome, picked, onPick. The focus effect.
      • ChoiceButton: props choice, letter, state, disabled, onPick.
    • Results: props quiz, answers, score, onRestart. A list of outcomes.

src/App.jsx 117 lines

The parent. It holds all the state and runs the effect that loads a set. It also picks which screen to show.

  • Six pieces of state. The score is not one of them: it is worked out from answers on every draw.
  • The effect depends on setId and tries. A new set, or Try again, loads the file again.
  • The ignore flag. Tap HTML then CSS quickly, and the late HTML answer is thrown away.
  • response.ok is checked, because a 404 does not make fetch fail by itself.
  • Three conditions decide the screen: error, loading, or a question. Then the results.
import { useEffect, useState } from 'react'
import SetPicker from './components/SetPicker.jsx'
import Progress from './components/Progress.jsx'
import QuestionCard from './components/QuestionCard.jsx'
import Results from './components/Results.jsx'
import './App.css'

// The three question sets. Each one is a file in public/questions/.
const SETS = [
  { id: 'html', name: 'HTML' },
  { id: 'css', name: 'CSS' },
  { id: 'js', name: 'JavaScript' },
]

export default function App() {
  // State: only what the app must remember between clicks.
  const [setId, setSetId] = useState('html')
  const [quiz, setQuiz] = useState(null)
  const [error, setError] = useState(null)
  const [tries, setTries] = useState(0)
  const [index, setIndex] = useState(0)
  const [answers, setAnswers] = useState([])

  // Effect: load the chosen set. The ignore flag throws away a late answer
  // if the student picks another set before this one arrives.
  useEffect(() => {
    let ignore = false
    fetch(`questions/${setId}.json`)
      .then((response) => {
        if (!response.ok) throw new Error(`The file ${setId}.json did not load (status ${response.status}).`)
        return response.json()
      })
      .then((data) => {
        if (!ignore) setQuiz(data)
      })
      .catch((err) => {
        if (!ignore) setError(err.message)
      })
    return () => {
      ignore = true
    }
  }, [setId, tries])

  function startOver() {
    setIndex(0)
    setAnswers([])
  }

  function pickSet(id) {
    if (id === setId) return
    setSetId(id)
    setQuiz(null)
    setError(null)
    startOver()
  }

  function tryAgain() {
    setError(null)
    setTries(tries + 1)
  }

  function pickChoice(choiceId) {
    const question = quiz.questions[index]
    setAnswers([...answers, { questionId: question.id, choiceId, correct: choiceId === question.answer }])
  }

  // Worked out from state on every draw, so it is never stored.
  const score = answers.filter((a) => a.correct).length
  const finished = quiz !== null && index === quiz.questions.length
  const question = quiz && !finished ? quiz.questions[index] : null
  const picked = question ? answers.find((a) => a.questionId === question.id) : null

  return (
    <div className="app">
      <header className="top">
        <p className="eyebrow">Checkpoint</p>
        <h1>Am I ready for React?</h1>
        <p className="lede">Pick a set. Answer 5 questions. See which outcomes to revisit.</p>
        <SetPicker sets={SETS} current={setId} onPick={pickSet} />
      </header>

      <main className="stage">
        {error && (
          <div className="panel error" role="alert">
            <p>{error}</p>
            <button type="button" className="btn" onClick={tryAgain}>Try again</button>
          </div>
        )}

        {!error && quiz === null && <p className="panel loading" role="status">Loading questions...</p>}

        {question && (
          <>
            <Progress questions={quiz.questions} answers={answers} index={index} />
            <QuestionCard
              key={question.id}
              question={question}
              number={index + 1}
              total={quiz.questions.length}
              outcome={quiz.outcomes.find((o) => o.id === question.outcome)}
              picked={picked ? picked.choiceId : null}
              onPick={pickChoice}
            />
            {picked && (
              <button type="button" className="btn primary next" onClick={() => setIndex(index + 1)}>
                {index + 1 === quiz.questions.length ? 'See my results' : 'Next question'}
              </button>
            )}
            <p className="score" aria-live="polite">{answers.length > 0 && `Score so far: ${score} of ${answers.length}`}</p>
          </>
        )}

        {finished && <Results quiz={quiz} answers={answers} score={score} onRestart={startOver} />}
      </main>
    </div>
  )
}

src/components/SetPicker.jsx 18 lines

The three set buttons. A list drawn with map, each with a key.

  • It owns no state. It gets the sets, the current one and onPick as props.
  • aria-pressed tells a screen reader which set is on.
// Props: the list of sets, the one in use, and what to do on a tap.
export default function SetPicker({ sets, current, onPick }) {
  return (
    <div className="picker" role="group" aria-label="Question set">
      {sets.map((set) => (
        <button
          key={set.id}
          type="button"
          className="chip"
          aria-pressed={set.id === current}
          onClick={() => onPick(set.id)}
        >
          {set.name}
        </button>
      ))}
    </div>
  )
}

src/components/Progress.jsx 18 lines

One dot per question.

  • Each dot's colour is worked out from props. The key is the question id, not the index.
// One dot per question: done right, done wrong, now, or still to come.
export default function Progress({ questions, answers, index }) {
  return (
    <ol className="progress" aria-label={`Question ${index + 1} of ${questions.length}`}>
      {questions.map((q, i) => {
        const answer = answers.find((a) => a.questionId === q.id)
        let state = 'todo'
        if (answer) state = answer.correct ? 'right' : 'wrong'
        else if (i === index) state = 'now'
        return (
          <li key={q.id} className={`dot ${state}`}>
            <span className="sr-only">{`Question ${i + 1}: ${state}`}</span>
          </li>
        )
      })}
    </ol>
  )
}

src/components/QuestionCard.jsx 54 lines

One question, its choices and the feedback.

  • App gives it key={question.id}, so each new question starts fresh.
  • A second, small effect moves keyboard focus to the new question. It needs no cleanup.
  • The feedback names the outcome and links to MDN, so students can check the answer themselves.
import { useEffect, useRef } from 'react'
import ChoiceButton from './ChoiceButton.jsx'

export default function QuestionCard({ question, number, total, outcome, picked, onPick }) {
  const heading = useRef(null)

  // After "Next question", move keyboard focus to the new question.
  useEffect(() => {
    if (number > 1) heading.current.focus()
  }, [number])

  const answered = picked !== null
  const right = picked === question.answer

  return (
    <section className="card" aria-labelledby={`q-${question.id}`}>
      <p className="meta">
        Question {number} of {total} <span className="tag">{outcome.id}</span>
      </p>
      <h2 id={`q-${question.id}`} ref={heading} tabIndex={-1}>{question.prompt}</h2>
      {question.code && <pre className="code"><code>{question.code}</code></pre>}

      <ul className="choices">
        {question.choices.map((choice, i) => {
          let state = 'open'
          if (answered && choice.id === question.answer) state = 'right'
          else if (answered && choice.id === picked) state = 'wrong'
          return (
            <ChoiceButton
              key={choice.id}
              choice={choice}
              letter={'ABCD'[i]}
              state={state}
              disabled={answered}
              onPick={onPick}
            />
          )
        })}
      </ul>

      <div aria-live="polite">
        {answered && (
          <div className={`feedback ${right ? 'good' : 'bad'}`}>
            <p className="verdict">{right ? 'Right.' : 'Not this time.'}</p>
            <p>{question.explain}</p>
            <p className="small">
              Outcome {outcome.id}: {outcome.text}. <a href={question.source} target="_blank" rel="noreferrer">Read more on MDN</a>
            </p>
          </div>
        )}
      </div>
    </section>
  )
}

src/components/ChoiceButton.jsx 13 lines

One answer button.

  • A tick or cross is added to the colour, so colour is never the only clue.
// A single answer. It shows a tick or a cross once the student has picked.
export default function ChoiceButton({ choice, letter, state, disabled, onPick }) {
  const mark = { right: ' ✓', wrong: ' ✗' }[state] || ''
  return (
    <li>
      <button type="button" className={`choice ${state}`} disabled={disabled} onClick={() => onPick(choice.id)}>
        <span className="letter" aria-hidden="true">{letter}</span>
        <span className="text">{choice.text}</span>
        <span className="mark" aria-hidden="true">{mark}</span>
      </button>
    </li>
  )
}

src/components/Results.jsx 30 lines

The end screen.

  • An outcome is met when every question on it was right. That uses filter and every.
  • The exit line is the bit I collect. Teachers get the same short line from every student.
// The end screen: the score, each outcome met or to revisit, and an exit line.
export default function Results({ quiz, answers, score, onRestart }) {
  const total = quiz.questions.length

  const outcomes = quiz.outcomes.map((outcome) => {
    const ids = quiz.questions.filter((q) => q.outcome === outcome.id).map((q) => q.id)
    const met = answers.filter((a) => ids.includes(a.questionId)).every((a) => a.correct)
    return { ...outcome, met }
  })
  const revisit = outcomes.filter((o) => !o.met).map((o) => o.id)
  const exitLine = `${quiz.title}: ${score}/${total}. Revisit: ${revisit.length ? revisit.join(', ') : 'none'}.`

  return (
    <section className="card results" aria-labelledby="results-title">
      <h2 id="results-title">You scored {score} of {total}</h2>
      <ul className="outcomes">
        {outcomes.map((o) => (
          <li key={o.id} className={o.met ? 'met' : 'revisit'}>
            <span className="tag">{o.id}</span>
            <span className="text">{o.text}</span>
            <strong>{o.met ? 'Met' : 'Revisit'}</strong>
          </li>
        ))}
      </ul>
      <p className="small">Copy this line into your class exit form:</p>
      <p className="exit">{exitLine}</p>
      <button type="button" className="btn primary" onClick={onRestart}>Do this set again</button>
    </section>
  )
}

# Simple workflow for deploying static content to GitHub Pages
name: Deploy static content to Pages

on:
  # Runs on pushes targeting the default branch
  push:
    branches: ['main']

  # Allows you to run this workflow manually from the Actions tab
  workflow_dispatch:

# Sets the GITHUB_TOKEN permissions to allow deployment to GitHub Pages
permissions:
  contents: read
  pages: write
  id-token: write

# Allow one concurrent deployment

  group: 'pages'
